\begin{frame}{C1 --- The boundary on geometric priors}
  \begin{itemize}
    \item \textbf{Question:} do spectral/topological/geometric invariants add
          information over taint for injection prioritization?
    \item \textbf{Measured on OWASP Benchmark 1.2 (2740 cases):} no.
    \item taint MRR $0.894$ vs.\ fused field $0.872$ ($p=5\times10^{-5}$);
          spectral $0.496$, topological $0.500$ (chance), curvature $0.376$.
    \item A flat servlet has no such structure; taint is near-optimal there.
    \item \textbf{Contribution:} a rigorous, falsifiable answer that redirects
          effort to where structure \emph{is} the signal.
  \end{itemize}
\end{frame}

\begin{frame}{C2 --- Authorization, mechanized and operational}
  \begin{itemize}
    \item $F:V\to\N$, auth edges $E_A=\{(u,v):F(u)<F(v)\}$; \texttt{no\_auth\_bounded}
          mechanized in Lean~4.
    \item BOLA/IDOR detector: ungated user-controlled resource access.
    \item \textbf{Z3 ownership witness}: $\mathrm{SAT}(\mathrm{owner}(id)\neq session)$.
    \item \texttt{login\_required} (authentication) does \emph{not} block BOLA ---
          only an object-level \texttt{owner} predicate does.
    \item \textbf{Real data (crAPI):} $9/9$ BOLA/BFLA recovered, recall $1.0$,
          precision $0.90$; missing-auth check scores $0$ (all are authenticated).
  \end{itemize}
\end{frame}

\begin{frame}{C3 --- Directed path homology}
  \begin{itemize}
    \item Symmetrization is lossy both ways: a 2-cycle collapses (missed), a
          filled triangle becomes a false cycle.
    \item GLMY recovers the concrete $\beta_1$ generator.
    \item \textbf{Measured (12 state graphs):} recall $1.0$ / precision $0.75$
          vs.\ undirected $0.5$ / $0.5$.
    \item GLMY $H_1$ = path-space holes $\supset$ feedback cycles: high recall,
          necessary-not-sufficient.
  \end{itemize}
\end{frame}

\begin{frame}{C4 --- The neuro-symbolic arbitration boundary}
  \begin{itemize}
    \item LLM \emph{proposes} a structured spec; Z3 \emph{disposes} it.
    \item \textbf{Measured (38 files, 29 findings):} detection/ranking agreement
          $0.974$ (LLM-independent); CWE changes $28\%$ (interpretation).
    \item Spec-mining (8 snippets): $0.5$ token recall, $0$ hallucination.
    \item \textbf{Contribution:} an architecture + its measurement, not a claim.
  \end{itemize}
\end{frame}

\begin{frame}{C5 --- Trust-topology analysis}
  \begin{itemize}
    \item Identity/permission graph: users, roles, trust policies, services.
    \item \textbf{Fiedler cut} surfaces trust-boundary crossings.
    \item \textbf{Forman--Ricci} isolates the privilege bridge (the attacker's pivot).
    \item Z3 resolves the role-assumption combinatorics.
  \end{itemize}
\end{frame}

\begin{frame}{Limitations (reported, not hidden)}
  \begin{itemize}
    \item BOLA/IDOR: one real API + a curated corpus, not a benchmark.
    \item The structural detector over-approximates (sanitized/parameterized).
    \item Z3 witnesses hold under the \emph{modeled} semantics, not end-to-end.
    \item GLMY $H_1$ is necessary, not sufficient.
  \end{itemize}
\end{frame}

\begin{frame}{Conclusion}
  \begin{itemize}
    \item The productive role of structural invariants in security is
          \emph{verification} of authorization, state, and trust properties ---
          not prioritization of injection sinks.
    \item All of it over one typed IR, emitting machine-checkable witnesses.
  \end{itemize}
