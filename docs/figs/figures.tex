% --- Figure 1: system pipeline --------------------------------------------
\begin{figure}[h]
  \centering
  \begin{tikzpicture}[
      box/.style={rectangle, draw=valen, thick, rounded corners, align=center,
                  font=\small, inner sep=4pt},
      arr/.style={-{Stealth[length=2mm]}, thick, valen},
    ]
    \node[box] (art) {artifact\\code / binary / API / IAM / agent};
    \node[box, right=6mm of art] (ir) {typed graph IR};
    \node[box, right=6mm of ir] (layers) {math layers\\spectral $\cdot$ topological $\cdot$ geometric $\cdot$ algebraic};
    \node[box, below=7mm of layers] (field) {structural field $V(x)$};
    \node[box, left=7mm of field] (agent) {neuro-symbolic\\agent (LLM)};
    \node[box, below=7mm of field] (verify) {Z3 $\cdot$ GLMY $\cdot$ Lean 4};
    \node[box, below=7mm of verify] (wit) {machine-checkable\\witnesses};
    \draw[arr] (art) -- (ir);
    \draw[arr] (ir) -- (layers);
    \draw[arr] (layers) -- (field);
    \draw[arr] (field) -- (agent);
    \draw[arr] (field) -- (verify);
    \draw[arr] (agent) -- (verify);
    \draw[arr] (verify) -- (wit);
  \end{tikzpicture}
  \caption{The pipeline: an artifact becomes one typed IR, enriched with the
  mathematical layers into a field $V(x)$; a neuro-symbolic loop proposes and the
  formal layer disposes, emitting only machine-checkable witnesses.}
\end{figure}

% --- Figure 3: directed vs symmetrized homology ---------------------------
\begin{figure}[h]
  \centering
  \begin{tikzpicture}
    \begin{axis}[
        ybar, bar width=0.28cm, width=0.9\linewidth, height=4.5cm,
        symbolic x coords={state machines, smart contracts},
        xtick=data, ymin=0, ymax=1.05, ylabel={recall},
        legend style={at={(0.5,1.02)}, anchor=south, legend columns=-1, font=\scriptsize},
        nodes near coords, every node near coord/.append style={font=\tiny},
        enlarge x limits=0.3,
      ]
      \addplot[fill=valen] coordinates {(state machines,1.00) (smart contracts,0.968)};
      \addplot[fill=gray!50]  coordinates {(state machines,0.50) (smart contracts,0.065)};
      \legend{GLMY directed, symmetrized}
    \end{axis}
  \end{tikzpicture}
  \caption{Directed (GLMY) path homology vs.\ symmetrized homology for destructive
  cycle detection. Symmetrization collapses the directed 2-cycle --- recall drops
  from $1.0$ to $0.5$ on state machines and from $0.97$ to $0.065$ on smart
  contracts.}
\end{figure}
